The Lead in Flint’s Water

In 2015, water samples were collected from homes across Flint, Michigan. Each drop represents a single test result. @cr-d3-plot

The EPA Action Level is 15 ppb. When more than 10% of samples exceed this, the water is legally considered unsafe. @cr-d3-plot

The 90th percentile for the original dataset was 15.8 ppb—exceeding the safety limit. @cr-d3-plot

However, specific high-lead samples were removed from the official report. @cr-d3-plot

With these samples gone, the new 90th percentile dropped to 11.4 ppb, making the water appear “safe.” @cr-d3-plot

d3 = require("d3@7")

// Load Data
payload = FileAttachment("data.json").json()
flintData = payload.data
stats = payload.stats

// Setup Dimensions - Parentheses are required for OJS object literals
width = 800
height = 500
margin = ({top: 40, right: 40, bottom: 60, left: 60})

// Scales
xScale = d3.scaleLinear()
  .domain([0, d3.max(flintData, d => d.lead)])
  .range([margin.left, width - margin.right])

yScale = d3.scaleLinear()
  .domain([0, d3.max(flintData, d => d.cumulative_count)])
  .range([height - margin.bottom, margin.top])

colorScale = d3.scaleLinear()
  .domain([0, 15, d3.max(flintData, d => d.lead)])
  .range(["#4682B4", "#ADD8E6", "#D2691E"])

// Main Plot Container
container = {
  const svg = d3.create("svg")
    .attr("viewBox", [0, 0, width, height])
    .style("background", "aliceblue")
    .style("width", "100%")
    .style("height", "auto");

  // Draw the droplets
  const drops = svg.append("g")
    .attr("class", "drops-layer")
    .selectAll("path")
    .data(flintData)
    .join("path")
    .attr("d", "M12 21.5C7 21.5 3 17.5 3 12.5C3 7.5 12 1.5 12 1.5C12 1.5 21 7.5 21 12.5C21 17.5 17 21.5 12 21.5Z")
    .attr("transform", d => `translate(${xScale(d.lead_bin)}, ${yScale(d.cumulative_count)}) scale(0.5)`)
    .attr("fill", d => colorScale(d.lead))
    .attr("opacity", 0.8)
    .attr("stroke", "white")
    .attr("stroke-width", 0.5);

  // EPA Line
  svg.append("line")
    .attr("class", "epa-line")
    .attr("x1", xScale(15)).attr("x2", xScale(15))
    .attr("y1", margin.top).attr("y2", height - margin.bottom)
    .attr("stroke", "red").attr("stroke-width", 3).attr("stroke-dasharray", "5,5").attr("opacity", 0);

  // 90th Percentile All
  svg.append("line")
    .attr("class", "p90-all")
    .attr("x1", xScale(stats.p90_all)).attr("x2", xScale(stats.p90_all))
    .attr("y1", margin.top).attr("y2", height - margin.bottom)
    .attr("stroke", "darkred").attr("stroke-width", 3).attr("opacity", 0);

  // 90th Percentile Filtered
  svg.append("line")
    .attr("class", "p90-filtered")
    .attr("x1", xScale(stats.p90_filtered)).attr("x2", xScale(stats.p90_filtered))
    .attr("y1", margin.top).attr("y2", height - margin.bottom)
    .attr("stroke", "darkgreen").attr("stroke-width", 3).attr("opacity", 0);

  return svg.node();
}

// Reactive Update Logic
{
  const trigger = typeof crTrigger !== "undefined" ? crTrigger : null;
  const svg = d3.select(container);
  
  // Animate lines
  svg.select(".epa-line").transition().duration(500).attr("opacity", trigger === "epa-line" ? 1 : 0);
  svg.select(".p90-all").transition().duration(500).attr("opacity", trigger === "percentile-90-all" ? 1 : 0);
  svg.select(".p90-filtered").transition().duration(500).attr("opacity", trigger === "percentile-90-filtered" ? 1 : 0);

  // Animate drops
  svg.select(".drops-layer").selectAll("path").transition().duration(500)
    .attr("opacity", d => {
      if (trigger === "highlight-removed") return d.removed ? 1 : 0.1;
      if (trigger === "percentile-90-filtered") return d.removed ? 0 : 0.8;
      return 0.8;
    })
    .attr("fill", d => (trigger === "highlight-removed" && d.removed) ? "salmon" : colorScale(d.lead));
}
