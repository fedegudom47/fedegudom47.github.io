Lead Concentration in Flint (2015)

This is standard markdown text.

This text is red using HTML.

1. Water Samples Collected
In 2015, water samples were collected from homes across Flint, Michigan. Each <strong>droplet</strong> here represents a sample.
2. The EPA Action Level
The EPA safety threshold is <strong>15 ppb</strong>. Any water testing above this line is considered unsafe.
3. The 90th Percentile
The key metric is the 90th percentile. For this dataset, it was dangerously high (roughly 18 ppb).
4. Identifying Outliers
However, several high-lead samples were identified for removal. We have highlighted them here in <span style="color: red; font-weight: bold;">RED</span>.
5. Removing the Data
These samples were removed from the official analysis. We now see only the "clean" dataset remaining.
6. The Resulting Drop
With those high-value points gone, the 90th percentile line <strong>slides down</strong> to 11.4 ppb—falling safely below the EPA action level.

raw_data = FileAttachment("https://raw.githubusercontent.com/rfordatascience/tidytuesday/main/data/2025/2025-11-04/flint_mdeq.csv").csv({typed: true})

plot_data = {
  const sorted = raw_data.sort((a, b) => a.sample - b.sample);
  const binCounts = {}; 

  return sorted.map(d => {
    const lead_bin = Math.floor(d.lead / 5) * 5;
    if (!binCounts[lead_bin]) binCounts[lead_bin] = 0;
    binCounts[lead_bin] += 1;
    
    const isRemoved = (d.lead2 === null || isNaN(d.lead2));

    return {
      ...d,
      lead_bin: lead_bin,
      cumulative_count: binCounts[lead_bin],
      is_removed: isRemoved
    };
  });
}

stats = {
  const all_leads = raw_data.map(d => d.lead).sort((a,b) => a - b);
  const filtered_leads = raw_data.filter(d => d.lead2 !== null).map(d => d.lead).sort((a,b) => a - b);
  
  return {
    p90_all: d3.quantile(all_leads, 0.9),
    p90_filtered: d3.quantile(filtered_leads, 0.9), // Should be ~11.4
    max_lead: d3.max(all_leads)
  }
}

// --- B. SCROLL TRACKER ---
viewof stepIndex = {
  const container = document.querySelector('#flint-text');
  const steps = container.querySelectorAll('.step');
  const debug = document.querySelector('#debug-step');

  container.value = 0; 
  
  const observer = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        const val = parseInt(entry.target.dataset.index);
        container.value = val;
        if(debug) debug.innerText = val; 
        container.dispatchEvent(new CustomEvent("input"));
      }
    });
  }, { threshold: 0.5 });
  
  steps.forEach(step => observer.observe(step));
  return container;
}

// --- C. CHART SETUP (Run Once) ---
canvas = {
  const width = 1000;
  const height = 700;
  // Increased margins to account for larger droplets at edges
  const margin = {top: 50, right: 50, bottom: 50, left: 50};

  const svg = d3.create("svg")
      .attr("viewBox", [0, 0, width, height])
      .style("width", "100%") 
      .style("height", "auto")
      .style("max-height", "95vh");

  const x = d3.scaleLinear()
      .domain([0, stats.max_lead]) 
      .range([margin.left, width - margin.right]);

  const y = d3.scaleLinear()
      .domain([0, 60]) 
      .range([height - margin.bottom, margin.top]);
      
  const color = d3.scaleLinear()
      .domain([0, 15, stats.max_lead])
      .range(["steelblue", "orange", "darkorange"]);

  // Axes
  svg.append("g")
      .attr("transform", `translate(0,${height - margin.bottom})`)
      .call(d3.axisBottom(x).tickSizeOuter(0))
      .call(g => g.selectAll("text").style("font-size", "14px"))
      .call(g => g.append("text").attr("x", width).attr("y", 40).text("Lead (ppb)").attr("fill", "black").attr("text-anchor", "end").style("font-size", "16px").style("font-weight", "bold"));
      
  svg.append("g")
      .attr("transform", `translate(${margin.left},0)`)
      .call(d3.axisLeft(y).tickSizeOuter(0))
      .call(g => g.selectAll("text").style("font-size", "14px"));

  // Legend
  const defs = svg.append("defs");
  const linearGradient = defs.append("linearGradient").attr("id", "legend-gradient");
  linearGradient.append("stop").attr("offset", "0%").attr("stop-color", "steelblue");
  linearGradient.append("stop").attr("offset", "15%").attr("stop-color", "orange"); 
  linearGradient.append("stop").attr("offset", "100%").attr("stop-color", "darkorange");

  const legend = svg.append("g").attr("transform", `translate(${width - 250}, ${margin.top})`);
  legend.append("text").attr("x", 0).attr("y", -10).text("Lead Concentration").style("font-weight", "bold");
  legend.append("rect").attr("width", 200).attr("height", 15).style("fill", "url(#legend-gradient)");
  const legendScale = d3.scaleLinear().domain([0, stats.max_lead]).range([0, 200]);
  legend.append("g").attr("transform", "translate(0,15)").call(d3.axisBottom(legendScale).ticks(5).tickSize(5));

  // Layers
  const dotLayer = svg.append("g").attr("class", "dots");
  const annotationLayer = svg.append("g").attr("class", "annotations");

  svg.node().scales = {x, y, color, margin, width}; // Passed 'width' for animation entrance
  svg.node().layers = {dotLayer, annotationLayer};
  
  return svg.node();
}

// --- D. UPDATE LOGIC (Runs on Scroll) ---
update = {
  const svg = d3.select(canvas);
  const {x, y, color, margin, width} = canvas.scales;
  const {dotLayer, annotationLayer} = canvas.layers;
  
  const t = svg.transition().duration(1200).ease(d3.easeCubicInOut);

  const dropletPath = "M0,-10 C-5,-10 -10,-5 -10,0 C-10,8 0,15 0,15 C0,15 10,8 10,0 C10,-5 5,-10 0,-10 Z";

  // 1. UPDATE DOTS
  dotLayer.selectAll("path")
    .data(plot_data)
    .join(
      enter => enter.append("path")
        .attr("d", dropletPath)
        .attr("transform", `translate(0, 0) scale(0)`)
        .attr("fill", d => color(d.lead))
        .attr("stroke", "white")
        .attr("stroke-width", 0.5)
        .call(enter => enter.transition(t)
            .delay((d, i) => i * 50) 
            .attr("transform", d => `translate(${x(d.lead_bin)}, ${y(d.cumulative_count)}) scale(1.5) rotate(180)`)
        ),
      
      update => update.transition(t)
        .attr("fill", d => {
          if (stepIndex === 3 && d.is_removed) return "#ff3333"; 
          return color(d.lead);
        })
        .attr("opacity", d => {
          if (stepIndex === 3 && !d.is_removed) return 0.1; 
          if (stepIndex >= 4 && d.is_removed) return 0;     
          return 1;
        })
        .attr("transform", d => `translate(${x(d.lead_bin)}, ${y(d.cumulative_count)}) scale(1.5)`)
    );

  // 2. UPDATE ANNOTATIONS
  const isFilteredState = (stepIndex >= 5);
  const currentP90 = isFilteredState ? stats.p90_filtered : stats.p90_all;
  const currentP90Color = isFilteredState ? "darkgreen" : "darkred";
  const currentP90Label = isFilteredState ? `New 90th Percentile: ${currentP90.toFixed(1)} ppb` : `90th Percentile: ${currentP90.toFixed(1)} ppb`;

  const activeAnnotations = [];
  
  if (stepIndex >= 1) {
    activeAnnotations.push({
      id: "epa-line", val: 15, color: "red", label: "EPA Action Level (15 ppb)", dash: "6,6"
    });
  }
  
  if (stepIndex >= 2) {
    activeAnnotations.push({
      id: "p90-line", val: currentP90, color: currentP90Color, label: currentP90Label, dash: "0"
    });
  }

  annotationLayer.selectAll(".ref-group")
    .data(activeAnnotations, d => d.id) 
    .join(
      enter => {
        const g = enter.append("g").attr("class", "ref-group");
        g.append("line")
          .attr("x1", d => x(d.val)).attr("x2", d => x(d.val))
          .attr("y1", y(60)).attr("y2", y(60))
          .attr("stroke", d => d.color).attr("stroke-width", 3).attr("stroke-dasharray", d => d.dash)
          .transition(t).attr("y2", y(0));
        g.append("text")
          .attr("x", d => x(d.val) + 10).attr("y", y(60) - 20)
          .text(d => d.label)
          .attr("fill", d => d.color).attr("font-weight", "bold").style("font-size", "16px")
          .attr("opacity", 0).transition(t).delay(500).attr("opacity", 1);
        return g;
      },
      update => {
        update.select("line").transition(t)
          .attr("x1", d => x(d.val)).attr("x2", d => x(d.val)).attr("stroke", d => d.color);
        update.select("text").transition(t)
          .attr("x", d => x(d.val) + 10).attr("fill", d => d.color)
          .tween("text", function(d) { this.textContent = d.label; });
        return update;
      },
      exit => exit.transition(t).style("opacity", 0).remove()
    );
}
